짐벌 회전각 θ_x 계산 (error_x → 각도)

# 프레임 가로폭 width, 바운딩 박스 중심 픽셀 center_x_pixel
THETA_MAX_DEG = 45.0   # 짐벌 허용 각도 (논문·판넬: ±45°)

def compute_gimbal_angle_deg(center_x_pixel: float, width: float) -> float:
    """정규화 center_x와 error_x로 짐벌 회전각(도)을 계산한다."""
    if width <= 0:
        return 0.0
    center_x = center_x_pixel / width          # [0.0, 1.0]
    error_x = center_x - 0.5                   # 화면 중앙(0.5) 대비 오차
    theta_max = THETA_MAX_DEG
    theta_x = error_x * 2 * theta_max          # θ_x = error_x × 2 × θ_max
    return max(-theta_max, min(theta_max, theta_x))

SMTP 실종 알림 (Missing 시 발송)

import smtplib
from email.mime.text import MIMEText

SMTP_SERVER = "smtp.gmail.com"
SMTP_PORT = 587
SENDER_EMAIL = "your_email@gmail.com"
SENDER_PASSWORD = "your_app_password"

WARNING_BODY = (
    "Warning: The monitored pet has disappeared from the camera frame"
)

def send_missing_alert_smtp(receiver_email: str) -> bool:
    """실종 판정 직후 SMTP로 경고 메일을 보낸다."""
    msg = MIMEText(WARNING_BODY, "plain", "utf-8")
    msg["Subject"] = "Pet monitoring alert"
    msg["From"] = SENDER_EMAIL
    msg["To"] = receiver_email

    try:
        with smtplib.SMTP(SMTP_SERVER, SMTP_PORT) as server:
            server.starttls()
            server.login(SENDER_EMAIL, SENDER_PASSWORD)
            server.sendmail(SENDER_EMAIL, [receiver_email], msg.as_string())
        return True
    except OSError:
        return False
